   \nu_{e} \\

   \nu_{\tau}
\end{pmatrix}
=
\begin{pmatrix}
   U_{e1} & U_{e2} & U_{e3} \\
   U_{\mu1} & U_{\mu2} & U_{\mu3} \\
   U_{\tau1} & U_{\tau2} & U_{\tau3}
\end{pmatrix}
\cdot
\begin{pmatrix}
   \nu_{1} \\
   \nu_{2}  \\
   \nu_{3}
\end{pmatrix}
$$

The PMNS matrix $U$ can be expressed as:
$$
U = \begin{pmatrix}
c_{12}c_{13} & s_{12}c_{13} & s_{13}e^{-i\delta_{CP}} \\

\end{pmatrix}
$$
